Script SQL de Inicialización (Supabase)

Pegá este código en el SQL Editor de tu Supabase para crear las 4 tablas principales.

-- ============================================================
-- SCRIPT DE INICIALIZACIÓN: TABLAS ENCUENTRO FAMILIAR
-- Ejecutalo en Supabase -> SQL Editor -> New Query -> Run
-- ============================================================

-- 1. Tabla de Familias
CREATE TABLE IF NOT EXISTS public.familias (
    id TEXT PRIMARY KEY,
    nombre TEXT NOT NULL,
    responsable TEXT,
    cuota_asignada NUMERIC DEFAULT 0,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Tabla de Asistentes / Integrantes (con control de entradas, salidas, roles y aportantes)
CREATE TABLE IF NOT EXISTS public.asistentes (
    id TEXT PRIMARY KEY,
    familia_id TEXT REFERENCES public.familias(id) ON DELETE CASCADE,
    nombre TEXT NOT NULL,
    categoria TEXT DEFAULT 'Adulto', -- 'Adulto', 'Criatura', 'Bebe'
    confirmado BOOLEAN DEFAULT TRUE,
    estado_flujo TEXT DEFAULT 'pendiente', -- 'en_predio', 'pendiente', 'retirado'
    hora_entrada TEXT,
    hora_salida TEXT,
    rol TEXT DEFAULT 'Participante',
    tareas TEXT,
    es_aportante BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Agregar columnas en caso de que las tablas ya existan
ALTER TABLE public.asistentes ADD COLUMN IF NOT EXISTS es_aportante BOOLEAN DEFAULT FALSE;

-- 3. Tabla de Gastos
CREATE TABLE IF NOT EXISTS public.gastos (
    id TEXT PRIMARY KEY,
    descripcion TEXT NOT NULL,
    categoria TEXT DEFAULT '1 de Enero',
    cantidad NUMERIC DEFAULT 1,
    costo_total NUMERIC DEFAULT 0,
    total_abonado NUMERIC DEFAULT 0,
    observaciones TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. Tabla de Aportes
CREATE TABLE IF NOT EXISTS public.aportes (
    id TEXT PRIMARY KEY,
    familia_id TEXT REFERENCES public.familias(id) ON DELETE CASCADE,
    persona_id TEXT,
    monto NUMERIC NOT NULL,
    metodo TEXT DEFAULT 'Transferencia Bancaria',
    fecha DATE DEFAULT CURRENT_DATE,
    nota TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE public.aportes ADD COLUMN IF NOT EXISTS persona_id TEXT;

-- 5. Habilitar Seguridad por Fila (RLS) abierta para sincronización de la app
ALTER TABLE public.familias ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.asistentes ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.gastos ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.aportes ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Acceso total anonimo familias" ON public.familias;
DROP POLICY IF EXISTS "Acceso total anonimo asistentes" ON public.asistentes;
DROP POLICY IF EXISTS "Acceso total anonimo gastos" ON public.gastos;
DROP POLICY IF EXISTS "Acceso total anonimo aportes" ON public.aportes;

CREATE POLICY "Acceso total anonimo familias" ON public.familias FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Acceso total anonimo asistentes" ON public.asistentes FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Acceso total anonimo gastos" ON public.gastos FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Acceso total anonimo aportes" ON public.aportes FOR ALL USING (true) WITH CHECK (true);
